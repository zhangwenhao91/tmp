import torch
import torch_npu
from tileops.backend import TensorSpec
from tileops.backends.ascend.builders.elementwise import build_add
from tileops.backends.ascend.runtime import support

npu = torch.device("npu")
for shape in [(32, 128), (4096, 128), (64, 256)]:
    artifact = support.vadd_artifact(torch.float32, shape[-1])
    spec = TensorSpec(npu, torch.float32, shape)
    kernel = build_add(spec, spec)          # 走 builder 的全部校验
    a = torch.randn(*shape, dtype=torch.float32, device=npu)
    b = torch.randn(*shape, dtype=torch.float32, device=npu)
    out = kernel(a, b)
    err = (out.cpu() - (a.cpu() + b.cpu())).abs().max().item()
    print(f"{shape} rows={a.numel() // shape[-1]} obj={artifact.path.name} "
          f"cv_mode={artifact.cv_mode} max_abs_err={err:.3e}", flush=True)
