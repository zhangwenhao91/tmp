# 1) 卸掉 CUDA 版 torch（遗留的 nvidia-* 包无害，可不管；想省磁盘见下方备注）
pip uninstall -y torch

# 2) 装 CPU 版（这次带上 trusted-host 绕过代理证书问题）
pip install torch==2.13.0 --index-url https://download.pytorch.org/whl/cpu \
  --trusted-host download.pytorch.org --trusted-host download-r2.pytorch.org



pip install torch==2.13.0+cpu -f https://mirrors.aliyun.com/pytorch-wheels/cpu/ \
  --trusted-host mirrors.aliyun.com



python -c "import torch,torch_npu; print(torch.__version__, torch_npu.__version__); \
x=torch.randn(3).npu(); torch.npu.synchronize(); print('NPU OK', x.sum().item())"
