# --- A) 真机 SOC 架构 ---
cat /usr/local/Ascend/driver/version.info 2>/dev/null | grep -iE "soc|product|chip|version"
ls /usr/local/Ascend/driver/tools/ 2>/dev/null
/usr/local/Ascend/driver/tools/npu-smi info 2>&1 | head -25

# --- B) 本次运行新建的 plog：先看 SOC 名，再看 aicore 异常细节 ---
ls -lt /root/ascend/log/run/plog/*.log 2>/dev/null | head -3
P=$(ls -t /root/ascend/log/run/plog/*.log 2>/dev/null | head -1)
echo "=== newest plog: $P"
grep -iE "soc|chip|product" "$P" 2>/dev/null | tail -8
echo "=== aicore/exception ==="
grep -iE "aicore|excep|fault|0x7bc87|0x507015|task" "$P" 2>/dev/null | grep -viE "INFO" | tail -40
ls -lt /root/ascend/log/debug/plog/*.log 2>/dev/null | head -3
D=$(ls -t /root/ascend/log/debug/plog/*.log 2>/dev/null | head -1)
grep -iE "aicore|0x7bc87|0x507015|pc=|addr" "$D" 2>/dev/null | grep -viE "INFO" | tail -40
