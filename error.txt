# 1) bisheng 的 arch/目标接口 + 它认识的 950 串
timeout 20 /data/pri/Ascend/9.2.0.B060/cann-9.2.0/tools/bisheng_compiler/bin/bisheng --version 2>&1 | head -5
timeout 20 /data/pri/Ascend/9.2.0.B060/cann-9.2.0/tools/bisheng_compiler/bin/bisheng --help 2>&1 | grep -iE "arch|aicore|cce|target|soc|depth" | head -30
strings /data/pri/Ascend/9.2.0.B060/cann-9.2.0/tools/bisheng_compiler/bin/bisheng 2>/dev/null | grep -iE "ascend950|950" | sort -u | head -20

# 2) 官方 950 cube 预编译算子的 ELF 长相（它就是我们在复刻的对象）
K=/data/pri/Ascend/9.2.0.B060/cann-9.2.0/opp/built-in/op_impl/ai_core/tbe/kernel/ascend950
ls $K/*.o 2>/dev/null | head
O=$(ls $K/*.o 2>/dev/null | head -1); echo "[$O]"
readelf -h "$O" 2>/dev/null | head -14
readelf -S "$O" 2>/dev/null | grep -iE "text|hacc|aicore|\.hc$" | head
strings "$O" 2>/dev/null | grep -iE "dav-|magic|aicore|kernel_type|high_perf" | head -15
